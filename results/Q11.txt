Q11 | 집계 | 판매 가능한 메뉴의 분류별 평균 가격을 구한다. AVG + GROUP BY.
SELECT category, ROUND(AVG(price_won), 2) AS average_price_won
FROM menu
WHERE is_available = 1
GROUP BY category
ORDER BY category ASC;
