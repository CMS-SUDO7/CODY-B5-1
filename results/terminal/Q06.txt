SELECT i.order_item_id, m.menu_name, i.quantity,
       i.unit_price_won, i.quantity * i.unit_price_won AS subtotal_won
FROM order_item AS i
INNER JOIN menu AS m ON m.menu_id = i.menu_id
ORDER BY i.order_item_id ASC;
