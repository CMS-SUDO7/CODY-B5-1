SELECT order_id, customer_id, ordered_at, status
FROM cafe_order
WHERE ordered_at >= '2026-10-01 00:00:00'
  AND ordered_at < '2026-11-01 00:00:00'
ORDER BY ordered_at DESC, order_id DESC
LIMIT 5;
