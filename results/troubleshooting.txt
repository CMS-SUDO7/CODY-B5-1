CASE 1: FK 검사가 꺼진 연결에서 없는 고객의 주문이 저장됨
PRAGMA foreign_keys = OFF;
실행 성공
PRAGMA foreign_keys;
결과: [(0,)]
SAVEPOINT invalid_order;
실행 성공
INSERT INTO cafe_order VALUES (99, 999, '2026-10-29 10:00:00', 'paid');
실행 성공 / 변경 행 수: 1
SELECT order_id, customer_id FROM cafe_order WHERE order_id = 99;
결과: [(99, 999)]
ROLLBACK TO invalid_order;
실행 성공
RELEASE invalid_order;
실행 성공
해결: 잘못된 입력을 되돌리고 트랜잭션 밖에서 FK 검사를 켬.
PRAGMA foreign_keys = ON;
실행 성공
PRAGMA foreign_keys;
결과: [(1,)]
SAVEPOINT invalid_order;
실행 성공
INSERT INTO cafe_order VALUES (99, 999, '2026-10-29 10:00:00', 'paid');
실제 오류: FOREIGN KEY constraint failed
ROLLBACK TO invalid_order;
실행 성공
RELEASE invalid_order;
실행 성공
SELECT COUNT(*) FROM cafe_order WHERE order_id = 99;
결과: [(0,)]
PRAGMA foreign_key_check;
결과: []

CASE 2: 이미 시작한 트랜잭션 안에서 다시 BEGIN 실행
BEGIN;
실행 성공
BEGIN TRANSACTION;
실제 오류: cannot start a transaction within a transaction
해결: 열린 작업을 ROLLBACK한 뒤 트랜잭션을 한 번만 시작함.
ROLLBACK;
실행 성공
PRAGMA foreign_keys = ON;
실행 성공
BEGIN TRANSACTION;
실행 성공
SELECT COUNT(*) FROM cafe_order;
결과: [(14,)]
COMMIT;
실행 성공

CASE 3: 초기화한 DB에 스키마 파일을 다시 실행
connection.executescript(schema_sql)
실제 오류: table customer already exists
ROLLBACK;
실행 성공
해결: 새 빈 DB에 01_schema.sql -> 02_seed.sql 순서로 실행함.
SELECT 'customer', COUNT(*) FROM customer UNION ALL SELECT 'menu', COUNT(*) FROM menu UNION ALL SELECT 'cafe_order', COUNT(*) FROM cafe_order UNION ALL SELECT 'order_item', COUNT(*) FROM order_item;
결과: [('customer', 10), ('menu', 10), ('cafe_order', 14), ('order_item', 23)]
PRAGMA integrity_check;
결과: [('ok',)]
PRAGMA foreign_key_check;
결과: []

CASE 4: executescript로 Q08을 실행했지만 결과 행을 얻지 못함
connection.executescript(q08_sql).fetchall()
결과: []
해결: 단일 SELECT는 execute로 실행하고 fetchall로 결과를 가져옴.
SELECT c.customer_name, o.order_id, m.menu_name, i.quantity
FROM customer AS c
INNER JOIN cafe_order AS o ON o.customer_id = c.customer_id
INNER JOIN order_item AS i ON i.order_id = o.order_id
INNER JOIN menu AS m ON m.menu_id = i.menu_id
WHERE c.customer_id = 1 AND o.status = 'paid'
ORDER BY o.order_id ASC, i.order_item_id ASC;
결과: [('김민수', 1, '아메리카노', 2), ('김민수', 1, '초코케이크', 1), ('김민수', 4, '콜드브루', 1), ('김민수', 4, '버터쿠키', 1)]
